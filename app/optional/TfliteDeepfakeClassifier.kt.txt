// thêm dependency  implementation("org.tensorflow:tensorflow-lite:2.14.0")
// và đặt mô hình vào app/src/main/assets/deepfake.tflite.
//
// Giả định về mô hình (chỉnh lại cho khớp mô hình thật của bạn):
//   - đầu vào  [1, 224, 224, 3] float32, giá trị 0..1 (RGB)
//   - đầu ra   [1, 1] = xác suất giả, hoặc [1, 2] = [thật, giả]
// Dự án KHÔNG kèm mô hình; cần mô hình đã huấn luyện (ví dụ trên FaceForensics++, DFDC) và đã kiểm
// tra độ chính xác trên video qua Zalo/Messenger thực tế (nén mạnh, ánh sáng kém).

package vn.scamshield.face

import android.content.Context
import android.graphics.Bitmap
import org.tensorflow.lite.Interpreter
import java.io.FileInputStream
import java.nio.ByteBuffer
import java.nio.ByteOrder
import java.nio.channels.FileChannel

class TfliteDeepfakeClassifier(context: Context, assetName: String = "deepfake.tflite") : DeepfakeClassifier {

    private val interpreter: Interpreter? = try {
        val fd = context.assets.openFd(assetName)
        val buf = FileInputStream(fd.fileDescriptor).channel
            .map(FileChannel.MapMode.READ_ONLY, fd.startOffset, fd.declaredLength)
        Interpreter(buf)
    } catch (e: Exception) {
        null
    }

    override val isAvailable: Boolean get() = interpreter != null

    override fun fakeProbability(face: Bitmap): Float? {
        val it = interpreter ?: return null
        val size = 224
        val scaled = Bitmap.createScaledBitmap(face, size, size, true)
        val input = ByteBuffer.allocateDirect(4 * size * size * 3).order(ByteOrder.nativeOrder())
        val px = IntArray(size * size)
        scaled.getPixels(px, 0, size, 0, 0, size, size)
        for (p in px) {
            input.putFloat(((p shr 16) and 0xFF) / 255f)
            input.putFloat(((p shr 8) and 0xFF) / 255f)
            input.putFloat((p and 0xFF) / 255f)
        }
        input.rewind()
        val outShape = it.getOutputTensor(0).shape()
        val n = outShape.last()
        val out = Array(1) { FloatArray(n) }
        it.run(input, out)
        return if (n == 1) out[0][0] else out[0][1]
    }
}
